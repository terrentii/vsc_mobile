> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** P2P-канал МЫС на Kotlin, совместимый с десктопом байт-в-байт: фраза → rendezvous → relay → CPace → Double Ratchet, текст и файлы.

**Architecture:** JVM-модуль `:p2p` (protocol, filetransfer, transport/rendezvous на OkHttp WebSocket, handshake, session, P2PService на корутинах) поверх `:crypto`; хранилище — интерфейсы `P2PStore`/`SessionStore`, реализация на vault в `:app`. **Tech Stack:** Kotlin 2.1.21, kotlinx-coroutines 1.10.2, OkHttp 4.12.0, Bouncy Castle 1.81 (BLAKE2b), lazysodium (ristretto255).

Спека: `docs/superpowers/specs/2026-10-22-p2p-relay.md`. Команды — из корня `vsc_mobile` в `nix develop`.

```kotlin
fun String.unhex(): ByteArray = ByteArray(length / 2) { substring(it * 2, it * 2 + 2).toInt(16).toByte() }
fun ByteArray.hex(): String = joinToString("") { "%02x".format(it) }
fun JSONObject.bytes(key: String): ByteArray = getString(key).unhex()

/** Дуплекс без сети: close() одной стороны обрывает recv() другой (как у десктопа). */
class InMemoryTransport private constructor(
    private val inbox: Channel<ByteArray>,
    private val outbox: Channel<ByteArray>,
) : Transport {
    override suspend fun send(data: ByteArray) {
        try { outbox.send(data) } catch (e: ClosedSendChannelException) { throw TransportException("send в закрытый транспорт") }
    }

    override suspend fun recv(): ByteArray =
        try { inbox.receive() } catch (e: ClosedReceiveChannelException) { throw TransportException("транспорт закрыт") }

    override suspend fun close() {
        outbox.close()
        inbox.close()
    }

    companion object {
        fun pair(): kotlin.Pair<InMemoryTransport, InMemoryTransport> {
            val ab = Channel<ByteArray>(Channel.UNLIMITED)
            val ba = Channel<ByteArray>(Channel.UNLIMITED)
            return InMemoryTransport(ba, ab) to InMemoryTransport(ab, ba)
        }
    }
}

/** P2PStore в памяти; ratchet хранится сериализованным — как в БД. */
class MemoryStore : P2PStore {
    class Stored(val id: Long, val direction: String, val body: ByteArray, var status: String, val kind: String, val filename: String?)
    class Conv(val id: Long, val roomId: ByteArray) {
        var prs: ByteArray? = null
        var ratchet: ByteArray? = null
        val messages = mutableListOf<Stored>()
    }

    private var nextId = 1L
    private val convs = mutableListOf<Conv>()

    @Synchronized fun conv(id: Long) = convs.first { it.id == id }
    @Synchronized fun messages(id: Long) = conv(id).messages.toList()

    @Synchronized override fun conversationFor(roomId: ByteArray): Long =
        (convs.find { it.roomId.contentEquals(roomId) } ?: Conv(nextId++, roomId).also { convs += it }).id

    @Synchronized override fun setPrs(conversationId: Long, prs: ByteArray) { conv(conversationId).prs = prs }

    @Synchronized override fun roomAndPrs(conversationId: Long) =
        conv(conversationId).let { c -> c.prs?.let { c.roomId to it } }

    @Synchronized private fun add(conv: Long, direction: String, body: ByteArray, status: String, kind: String, filename: String?): Long {
        val m = Stored(nextId++, direction, body, status, kind, filename)
        conv(conv).messages += m
        return m.id
    }

    override fun session(conversationId: Long): SessionStore = object : SessionStore {
        val store = this@MemoryStore
        override fun loadRatchet(): RatchetState? = synchronized(store) { conv(conversationId).ratchet?.let(::deserializeState) }
        override fun saveRatchet(state: RatchetState) = synchronized(store) { conv(conversationId).ratchet = serializeState(state) }
        override fun addOutgoing(body: ByteArray, status: String, kind: String, filename: String?, mimeType: String?) =
            add(conversationId, "out", body, status, kind, filename)
        override fun setStatus(messageId: Long, status: String) =
            synchronized(store) { conv(conversationId).messages.first { it.id == messageId }.status = status }
        override fun receive(body: ByteArray, state: RatchetState, kind: String, filename: String?, mimeType: String?): Long =
            synchronized(store) { saveRatchet(state); add(conversationId, "in", body, "received", kind, filename) }
    }
}

/** Ждать условия (сеть/корутины асинхронны). */
suspend fun eventually(timeoutMs: Long = 10_000, check: () -> Boolean) = withTimeout(timeoutMs) {
    while (!check()) delay(20)
}
```

```kotlin
package ru.mobile.mys.p2p

import java.io.ByteArrayOutputStream
import java.text.Normalizer
import org.bouncycastle.crypto.digests.Blake2bDigest
import ru.mobile.mys.crypto.hkdf

/*
 * Wire-протокол МЫС — порт mys_decentralized/protocol.py (байт-в-байт).
 * Кадр: u8 type | u8 flags | u32 length | payload (big-endian). Сервер видит
 * только room_id и непрозрачные кадры.
 */

typealias Candidate = kotlin.Pair<String, Int>

private val NO_SALT = ByteArray(0)

/** BLAKE2b с personalization; Python дополняет `person` нулями до 16 байт. */
internal fun blake2b(data: ByteArray, size: Int, person: String): ByteArray {
    val d = Blake2bDigest(null, size, null, person.toByteArray().copyOf(16))
    d.update(data, 0, data.size)
    return ByteArray(size).also { d.doFinal(it, 0) }
}

/** Фраза → (room_id, prs): room_id уходит серверу, prs (вход CPace) — нет. */
fun deriveRoomParams(phrase: String): kotlin.Pair<ByteArray, ByteArray> {
    // ponytail: trim() ≈ Python str.strip(); расходятся лишь на экзотических пробелах
    // (после NFKC почти все сводятся к U+0020). Векторы покрывают обычные случаи.
    val norm = Normalizer.normalize(phrase, Normalizer.Form.NFKC).trim().toByteArray()
    val seed = blake2b(norm, 32, "mys-phrase")
    return hkdf(seed, 32, NO_SALT, "mys-room-id".toByteArray()) to hkdf(seed, 32, NO_SALT, "mys-pake-prs".toByteArray())
}

enum class MsgType(val code: Int) {
    HELLO(1), PAIR(2), PUNCH(3), PUNCH_ACK(4), RELAY(5), PAKE(6), CONFIRM(7), DATA(8), PEER_LEFT(9);

    companion object {
        fun of(code: Int): MsgType? = entries.firstOrNull { it.code == code }
    }
}

/** Порядок важен: ordinal = байт на проводе. INITIATOR — Alice в ratchet. */
enum class Role { INITIATOR, RESPONDER }

internal class Writer {
    private val out = ByteArrayOutputStream()
    fun u8(v: Int) = apply { require(v in 0..0xFF); out.write(v) }
    fun u16(v: Int) = apply { require(v in 0..0xFFFF); out.write(v ushr 8); out.write(v) }
    fun u32(v: Int) = apply { for (s in intArrayOf(24, 16, 8, 0)) out.write(v ushr s) }
    fun u64(v: Long) = apply { for (s in 56 downTo 0 step 8) out.write((v ushr s).toInt()) }
    fun bytes(b: ByteArray) = apply { out.write(b) }
    fun var8(b: ByteArray) = u8(b.size).bytes(b)
    fun var16(b: ByteArray) = u16(b.size).bytes(b)
    fun candidates(c: List<Candidate>) = apply {
        u8(c.size)
        for ((host, port) in c) var16(host.toByteArray()).u16(port)
    }
    fun done(): ByteArray = out.toByteArray()
}

internal class Reader(private val b: ByteArray) {
    private var pos = 0
    private fun need(n: Int) { if (n < 0 || pos + n > b.size) throw TransportException("усечённое поле") }
    fun u8(): Int { need(1); return b[pos++].toInt() and 0xff }
    fun u16(): Int = (u8() shl 8) or u8()
    fun u32(): Long = (u16().toLong() shl 16) or u16().toLong()
    fun u64(): Long = (u32() shl 32) or u32()
    fun take(n: Int): ByteArray { need(n); return b.copyOfRange(pos, pos + n).also { pos += n } }
    fun rest(): ByteArray = take(b.size - pos)
    fun var8() = take(u8())
    fun var16() = take(u16())
    fun candidates(): List<Candidate> = List(u8()) { var16().decodeToString() to u16() }
}

sealed class Msg(val type: MsgType) {
    abstract fun payload(): ByteArray

    class Hello(val roomId: ByteArray, val candidates: List<Candidate> = emptyList()) : Msg(MsgType.HELLO) {
        override fun payload() = Writer().var16(roomId).candidates(candidates).done()
    }

    class Pair(val role: Role, val peerCandidates: List<Candidate> = emptyList()) : Msg(MsgType.PAIR) {
        override fun payload() = Writer().u8(role.ordinal).candidates(peerCandidates).done()
    }

    /** CPace-точка Y — открыта по дизайну PAKE. */
    class Pake(val y: ByteArray, val ad: ByteArray = ByteArray(0)) : Msg(MsgType.PAKE) {
        override fun payload() = Writer().var16(y).var16(ad).done()
    }

    class Confirm(val mac: ByteArray) : Msg(MsgType.CONFIRM) {
        override fun payload() = mac
    }

    class Data(val sealed: ByteArray) : Msg(MsgType.DATA) {
        override fun payload() = sealed
    }

    /** Кадр для пира целиком; сервер пересылает его, не читая. */
    class Relay(val inner: ByteArray) : Msg(MsgType.RELAY) {
        override fun payload() = inner
    }

    object Punch : Msg(MsgType.PUNCH) { override fun payload() = ByteArray(0) }
    object PunchAck : Msg(MsgType.PUNCH_ACK) { override fun payload() = ByteArray(0) }

    /** Сервер → оставшемуся: второй участник отключился. */
    object PeerLeft : Msg(MsgType.PEER_LEFT) { override fun payload() = ByteArray(0) }
}

class Frame(val type: MsgType, val flags: Int, val payload: ByteArray, val consumed: Int)

fun encodeFrame(type: MsgType, payload: ByteArray, flags: Int = 0): ByteArray =
    Writer().u8(type.code).u8(flags and 0xFF).u32(payload.size).bytes(payload).done()

/** Разобрать первый кадр из [buf]; `consumed` — его длина (хвост — следующий кадр). */
fun decodeFrame(buf: ByteArray): Frame {
    if (buf.size < 6) throw TransportException("усечённый заголовок кадра")
    val r = Reader(buf)
    val rawType = r.u8()
    val flags = r.u8()
    val length = r.u32()
    if (6 + length > buf.size) throw TransportException("неполный payload кадра")
    val type = MsgType.of(rawType) ?: throw TransportException("неизвестный тип кадра $rawType")
    return Frame(type, flags, r.take(length.toInt()), 6 + length.toInt())
}

fun encodeMessage(msg: Msg): ByteArray = encodeFrame(msg.type, msg.payload())

fun decodeMessage(buf: ByteArray): Msg {
    val f = decodeFrame(buf)
    val r = Reader(f.payload)
    return when (f.type) {
        MsgType.HELLO -> Msg.Hello(r.var16(), r.candidates())
        MsgType.PAIR -> {
            if (f.payload.isEmpty()) throw TransportException("пустой PAIR")

        MsgType.CONFIRM -> Msg.Confirm(f.payload)
        MsgType.DATA -> Msg.Data(f.payload)
        MsgType.RELAY -> Msg.Relay(f.payload)
        MsgType.PUNCH -> Msg.Punch
        MsgType.PUNCH_ACK -> Msg.PunchAck
        MsgType.PEER_LEFT -> Msg.PeerLeft
    }
```

```kotlin
package ru.mobile.mys.p2p

/*
 * Формат содержимого внутри envelope.seal — порт mys_decentralized/filetransfer.py.
 * Первый байт plaintext — kind; wire-кадры не меняются.

const val KIND_TEXT: Byte = 0x00
const val KIND_FILE_META: Byte = 0x01
const val KIND_FILE_CHUNK: Byte = 0x02

/** 64 КиБ: влезает и в лимит сообщения WebRTC (спека §2). Получатель берёт размер из META. */
const val CHUNK_SIZE = 64 * 1024
const val MAX_FILE_SIZE = 50 * 1024 * 1024
const val TRANSFER_ID_LEN = 16

private val IMAGE_EXTS = setOf("png", "jpg", "jpeg", "gif", "webp", "bmp", "tiff", "tif")

/** "image" — для инлайн-превью, иначе "file". */
fun kindForFilename(filename: String): String =
    if (filename.substringAfterLast('.', "").lowercase() in IMAGE_EXTS) "image" else "file"

class FileMeta(
    val transferId: ByteArray, val totalSize: Long, val chunkSize: Int, val chunkCount: Int,
    val sha256: ByteArray, val mimeType: String, val filename: String,
)

class FileChunk(val transferId: ByteArray, val index: Int, val data: ByteArray)

private const val META_FIXED = TRANSFER_ID_LEN + 8 + 4 + 4 + 32

fun encodeFileMeta(m: FileMeta): ByteArray {
    require(m.transferId.size == TRANSFER_ID_LEN && m.sha256.size == 32)
    return Writer().bytes(m.transferId).u64(m.totalSize).u32(m.chunkSize).u32(m.chunkCount)
        .bytes(m.sha256).var8(m.mimeType.toByteArray()).var16(m.filename.toByteArray()).done()
}

fun decodeFileMeta(buf: ByteArray): FileMeta {
    if (buf.size < META_FIXED) throw TransportException("усечённый заголовок FILE_META")
    val r = Reader(buf)
    return FileMeta(r.take(TRANSFER_ID_LEN), r.u64(), r.u32().toInt(), r.u32().toInt(), r.take(32),
        r.var8().decodeToString(), r.var16().decodeToString())
}

fun encodeFileChunk(c: FileChunk): ByteArray {
    require(c.transferId.size == TRANSFER_ID_LEN)
    return Writer().bytes(c.transferId).u32(c.index).bytes(c.data).done()
}

fun decodeFileChunk(buf: ByteArray): FileChunk {
    if (buf.size < TRANSFER_ID_LEN + 4) throw TransportException("усечённый заголовок FILE_CHUNK")
    val r = Reader(buf)
    return FileChunk(r.take(TRANSFER_ID_LEN), r.u32().toInt(), r.rest())
}

fun splitChunks(data: ByteArray, size: Int = CHUNK_SIZE): List<ByteArray> =
    (data.indices step size).map { data.copyOfRange(it, minOf(it + size, data.size)) }

fun validateMeta(m: FileMeta) {
    require(m.totalSize <= MAX_FILE_SIZE) { "файл превышает лимит $MAX_FILE_SIZE байт" }
    require(m.chunkSize > 0) { "chunk_size должен быть положительным" }
    val expected = (m.totalSize + m.chunkSize - 1) / m.chunkSize
    require(m.chunkCount.toLong() == expected) { "chunk_count=${m.chunkCount} не соответствует ожидаемому $expected" }
}
```

```kotlin
import okio.ByteString

/** WebSocket (OkHttp) как очередь бинарных сообщений: одно сообщение = один кадр. */
class WsConnection private constructor() {
    private lateinit var ws: WebSocket
    private val inbox = Channel<ByteArray>(Channel.UNLIMITED)

    fun send(data: ByteArray) {
        if (!ws.send(data.toByteString())) throw TransportException("WebSocket закрыт")
    }

    suspend fun recv(): ByteArray =
        try { inbox.receive() } catch (e: ClosedReceiveChannelException) { throw TransportException("WebSocket закрыт") }

    fun close() {
        ws.close(1000, null)
        inbox.close()
    }

    companion object {
        suspend fun connect(client: OkHttpClient, url: String): WsConnection = suspendCancellableCoroutine { cont ->
            val conn = WsConnection()
            val listener = object : WebSocketListener() {
                override fun onOpen(webSocket: WebSocket, response: Response) = cont.resume(conn)
                override fun onMessage(webSocket: WebSocket, bytes: ByteString) { conn.inbox.trySend(bytes.toByteArray()) }
                override fun onClosing(webSocket: WebSocket, code: Int, reason: String) {
                    webSocket.close(1000, null)
                    conn.inbox.close()
                }
                override fun onClosed(webSocket: WebSocket, code: Int, reason: String) { conn.inbox.close() }
                override fun onFailure(webSocket: WebSocket, t: Throwable, response: Response?) {
                    if (cont.isActive) cont.resumeWithException(RendezvousException("не удалось подключиться к rendezvous-серверу", t))
                    conn.inbox.close()
                }
            }
            conn.ws = client.newWebSocket(Request.Builder().url(url).build(), listener)
            cont.invokeOnCancellation { conn.ws.cancel() }
        }
    }
}

/** Кадры к пиру — в `RELAY{кадр}` через rendezvous-сервер, который их не читает. */
class RelayTransport(private val ws: WsConnection) : Transport {
    override suspend fun send(data: ByteArray) = ws.send(encodeMessage(Msg.Relay(data)))

    override suspend fun recv(): ByteArray = when (val m = decodeMessage(ws.recv())) {
        is Msg.Relay -> m.inner
        // Сервер сигналит уход пира отдельным кадром — для сессии это обрыв канала.
        is Msg.PeerLeft -> throw TransportException("пир покинул комнату")
        else -> throw TransportException("ожидался RELAY-кадр")
    }

    override suspend fun close() = ws.close()
}

class Rendezvous(val role: Role, val peerCandidates: List<Candidate>, val ws: WsConnection)

/** Войти в комнату и дождаться пары. Кандидатов не анонсируем: прямой путь — WebRTC (под-проект 6). */
suspend fun joinRendezvous(client: OkHttpClient, url: String, roomId: ByteArray, timeoutMs: Long): Rendezvous {
    val ws = WsConnection.connect(client, url)
    try {
        ws.send(encodeMessage(Msg.Hello(roomId)))
        val first = withTimeoutOrNull(timeoutMs) { ws.recv() }
            ?: throw PeerUnavailable("собеседник не вышел на связь за отведённое время")
        val pair = decodeMessage(first) as? Msg.Pair ?: throw RendezvousException("ожидался PAIR от сервера")
        return Rendezvous(pair.role, pair.peerCandidates, ws)
    } catch (e: TransportException) {
        ws.close()
        throw RendezvousException("соединение с rendezvous-сервером оборвалось", e)
    } catch (e: Throwable) {
        ws.close()
        throw e
    }
}
```

```kotlin
package ru.mobile.mys.p2p

import java.security.MessageDigest
import ru.mobile.mys.crypto.KeyPair
import ru.mobile.mys.crypto.RatchetState
import ru.mobile.mys.crypto.Ristretto
import ru.mobile.mys.crypto.RistrettoException

import ru.mobile.mys.crypto.cpaceFinish
import ru.mobile.mys.crypto.cpaceMsg
import ru.mobile.mys.crypto.hkdf
import ru.mobile.mys.crypto.hmacSha256
import ru.mobile.mys.crypto.ratchetInitAlice
import ru.mobile.mys.crypto.ratchetInitBob
import ru.mobile.mys.crypto.x25519KeypairFromSeed

/*
 * Хендшейк — порт mys_decentralized/handshake.py: обмен PAKE{Y}, ISK из CPace,
 * обмен CONFIRM{HMAC(ISK, "mys-confirm" ‖ sort(Ya, Yb))}, затем sk и стартовый DH
 * Боба из ISK и ratchet по роли.
 */

class HandshakeResult(val sk: ByteArray, val ratchetState: RatchetState, val isk: ByteArray)

private val NONE = ByteArray(0)

fun deriveSid(prs: ByteArray): ByteArray = blake2b(prs, 16, "mys-cpace-sid")

fun confirmMac(isk: ByteArray, myY: ByteArray, peerY: ByteArray): ByteArray {
    val (low, high) = if (compareUnsigned(myY, peerY) <= 0) myY to peerY else peerY to myY
    return hmacSha256(isk, "mys-confirm".toByteArray() + low + high)
}

/** (sk — корень ratchet, стартовая DH-пара Боба). */
fun deriveSessionKeys(isk: ByteArray): kotlin.Pair<ByteArray, KeyPair> =
    hkdf(isk, 32, NONE, "mys-ratchet-root".toByteArray()) to
        x25519KeypairFromSeed(hkdf(isk, 32, NONE, "mys-ratchet-bob-dh".toByteArray()))

suspend fun handshake(transport: Transport, prs: ByteArray, role: Role, r: Ristretto): HandshakeResult {
    val (state, myY) = cpaceMsg(r, prs, deriveSid(prs))
    transport.send(encodeMessage(Msg.Pake(myY)))
    val peer = decodeMessage(transport.recv()) as? Msg.Pake ?: throw PakeException("ожидался PAKE на шаге обмена точками")
    val isk = try {
        cpaceFinish(r, state, peer.y)
    } catch (e: RistrettoException) {
        throw PakeException("невалидная точка пира", e)
    }
    val mac = confirmMac(isk, myY, peer.y)
    transport.send(encodeMessage(Msg.Confirm(mac)))
    val confirm = decodeMessage(transport.recv()) as? Msg.Confirm ?: throw PakeException("ожидался CONFIRM")
    if (!MessageDigest.isEqual(confirm.mac, mac)) throw PakeException("key-confirmation не сошёлся: неверная фраза или MITM")
    val (sk, bob) = deriveSessionKeys(isk)
    val ratchet = if (role == Role.INITIATOR) ratchetInitAlice(sk, bob.second) else ratchetInitBob(sk, bob)
    return HandshakeResult(sk, ratchet, isk)
}
```

```kotlin
    private val mutex = Mutex()
    private var recvJob: Job? = null
    private val outbox = ArrayDeque<kotlin.Pair<Long, ByteArray>>()
    private val pendingFiles = HashMap<ByteBuffer, PendingFile>()

    /** Есть ли отправляющая цепочка (у RESPONDER — после первого входящего). */
    val canSend: Boolean get() = state.cks != null

    fun start(scope: CoroutineScope) {
        if (recvJob == null) recvJob = scope.launch { recvLoop() }
    }

    suspend fun send(text: String) {
        val data = text.toByteArray()
        if (data.isEmpty()) return // пустой payload зарезервирован под prime
        val payload = byteArrayOf(KIND_TEXT) + data
        mutex.withLock {
            if (!canSend) {
                outbox.addLast(store.addOutgoing(data, "pending") to payload)
                return
            }
            sealSaveSend(payload)
            store.addOutgoing(data, "sent")
        }
    }

    /** META + чанки; требует [canSend] (файл в очередь не ставится — как в десктопе). */
    suspend fun sendFile(filename: String, mimeType: String, data: ByteArray): Long {
        require(data.size <= MAX_FILE_SIZE) { "файл превышает лимит $MAX_FILE_SIZE байт" }
        return mutex.withLock {
            check(canSend) { "канал ещё не готов для отправки файла (примите первое сообщение)" }
            val id = ByteArray(TRANSFER_ID_LEN).also { SecureRandom().nextBytes(it) }
            val chunks = splitChunks(data)
            val sha = MessageDigest.getInstance("SHA-256").digest(data)
            sealSaveSend(byteArrayOf(KIND_FILE_META) + encodeFileMeta(FileMeta(id, data.size.toLong(), CHUNK_SIZE, chunks.size, sha, mimeType, filename)))
            chunks.forEachIndexed { i, c -> sealSaveSend(byteArrayOf(KIND_FILE_CHUNK) + encodeFileChunk(FileChunk(id, i, c))) }
            store.addOutgoing(data, "sent", kindForFilename(filename), filename, mimeType)
        }
    }

    /** INITIATOR после подключения: пустой кадр даёт RESPONDER'у отправляющую цепочку. */
    suspend fun sendPrime() = mutex.withLock { if (canSend) sealSaveSend(ByteArray(0)) }

    /** Состояние сохраняется ДО отправки: упади процесс после send — ключ цепочки не переиспользуется. */
    private suspend fun sealSaveSend(payload: ByteArray) {
        val sealed = seal(state, tk, payload)
        store.saveRatchet(state)
        transport.send(encodeMessage(Msg.Data(sealed)))
    }

    private suspend fun flushOutboxLocked() {
        while (outbox.isNotEmpty() && canSend) {
            val (id, payload) = outbox.removeFirst()
            sealSaveSend(payload)
            store.setStatus(id, "sent")
        }
    }

    private suspend fun recvLoop() {
        try {
            while (true) handleFrame(transport.recv())
        } catch (e: TransportException) {
            // Обрыв сам по себе (close() отменяет задачу и сюда не попадает) — пир пропал.
            runCatching { transport.close() }
            runCatching { onDisconnect() }
        }
    }

    private suspend fun handleFrame(frame: ByteArray) {
        val msg = try { decodeMessage(frame) } catch (e: TransportException) { return }
        if (msg !is Msg.Data) return
        var receivedId: Long? = null
        mutex.withLock {
            val plaintext = try { open(state, tk, msg.sealed) } catch (e: Exception) { return } // ratchet не двинулся
            val incoming = if (plaintext.isEmpty()) null else runCatching { dispatch(plaintext) }.getOrNull()
            if (incoming == null) store.saveRatchet(state) // prime / промежуточный кадр файла
            else receivedId = store.receive(incoming.body, state, incoming.kind, incoming.filename, incoming.mimeType)
            flushOutboxLocked()
        }
        receivedId?.let(onMessage)
    }

    private fun dispatch(plaintext: ByteArray): Incoming? {
        val body = plaintext.copyOfRange(1, plaintext.size)
        return when (plaintext[0]) {
            KIND_TEXT -> Incoming("text", null, null, body)
            KIND_FILE_META -> decodeFileMeta(body).let { validateMeta(it); entry(it.transferId).meta = it; complete(it.transferId) }
            KIND_FILE_CHUNK -> decodeFileChunk(body).let { entry(it.transferId).chunks[it.index] = it.data; complete(it.transferId) }
            else -> null // неизвестный kind (напр. сигналинг WebRTC новых версий) — тихо отбросить
        }
    }

    private fun entry(id: ByteArray) = pendingFiles.getOrPut(ByteBuffer.wrap(id)) { PendingFile() }

    private fun complete(id: ByteArray): Incoming? {
        val key = ByteBuffer.wrap(id)
        val e = pendingFiles[key] ?: return null
        val meta = e.meta ?: return null
        if (e.chunks.size < meta.chunkCount) return null
        val out = ByteArrayOutputStream(meta.totalSize.toInt())
        for (i in 0 until meta.chunkCount) out.write(e.chunks[i] ?: return null)
        pendingFiles.remove(key)
        val full = out.toByteArray()
        if (!MessageDigest.isEqual(MessageDigest.getInstance("SHA-256").digest(full), meta.sha256)) return null
        return Incoming(kindForFilename(meta.filename), meta.filename, meta.mimeType, full)
    }

    suspend fun close() {
        recvJob?.cancelAndJoin()
        recvJob = null
        pendingFiles.clear()
        transport.close()
    }
}

/** Сессия беседы: ratchet возобновляется из хранилища, иначе сеется из хендшейка. */
fun openSession(
    store: SessionStore,
    transport: Transport,
    sk: ByteArray,
    seedState: RatchetState,
    onMessage: (Long) -> Unit = {},
    onDisconnect: () -> Unit = {},
): Session {
    val state = store.loadRatchet() ?: seedState.also { store.saveRatchet(it) }
    return Session(transport, state, deriveTransformKey(sk), store, onMessage, onDisconnect)
}
```

```kotlin
package ru.mobile.mys.p2p

import java.util.concurrent.ConcurrentHashMap
import kotlinx.coroutines.CancellationException
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.SupervisorJob
import kotlinx.coroutines.cancel

import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.withTimeoutOrNull
import okhttp3.OkHttpClient
import ru.mobile.mys.crypto.Ristretto

sealed interface P2PEvent {
    class Message(val conversationId: Long, val messageId: Long) : P2PEvent
    class State(val conversationId: Long, val connected: Boolean) : P2PEvent
    class Error(val conversationId: Long?, val error: Throwable) : P2PEvent
}

/**
 * Оркестратор P2P — порт mys_decentralized/service.py: rendezvous → relay →
 * хендшейк → сессия, активные сессии по conversation_id, события для UI.
 */
class P2PService(
    private val store: P2PStore,
    private val rendezvousUrl: String,
    private val ristretto: Ristretto,
    private val http: OkHttpClient = OkHttpClient(),
    private val connectTimeoutMs: Long = 300_000,
    private val handshakeTimeoutMs: Long = 30_000,
) {
    private val scope = CoroutineScope(SupervisorJob() + Dispatchers.IO)
    private val sessions = ConcurrentHashMap<Long, Session>()
    private val roles = ConcurrentHashMap<Long, Role>()
    private val _events = MutableSharedFlow<P2PEvent>(extraBufferCapacity = 256, onBufferOverflow = BufferOverflow.DROP_OLDEST)

    /** Сообщения уже в хранилище — события лишь будят UI. */
    val events: SharedFlow<P2PEvent> = _events

    fun hasSession(conversationId: Long) = sessions.containsKey(conversationId)

    fun roleOf(conversationId: Long): Role? = roles[conversationId]

    /** Беседа по фразе без сети — сразу показать окно ожидания пира. */
    fun resolveConversation(phrase: String): Long = store.conversationFor(deriveRoomParams(phrase).first)

    /** Поднять канал по фразе; ждёт пира до [timeoutMs]. */
    suspend fun startSession(phrase: String, timeoutMs: Long = connectTimeoutMs): Long {
        val (roomId, prs) = deriveRoomParams(phrase)
        val id = store.conversationFor(roomId)
        store.setPrs(id, prs) // до сети: даже при таймауте канал реконнектится без фразы
        connect(id, roomId, prs, timeoutMs)
        return id
    }

    /** «Выйти на связь» без повторного ввода фразы. */
    suspend fun reconnect(conversationId: Long, timeoutMs: Long = connectTimeoutMs) {
        val (roomId, prs) = store.roomAndPrs(conversationId) ?: throw IllegalStateException(
            "для этого канала нет сохранённой фразы — откройте его заново через «+ Новый канал» с той же фразой",
        )
        connect(conversationId, roomId, prs, timeoutMs)
    }

    suspend fun send(conversationId: Long, text: String) = session(conversationId).send(text)

    suspend fun sendFile(conversationId: Long, filename: String, mimeType: String, data: ByteArray): Long =
        session(conversationId).sendFile(filename, mimeType, data)

    suspend fun stopSession(conversationId: Long) {
        roles.remove(conversationId)
        sessions.remove(conversationId)?.let {
            it.close()
            _events.tryEmit(P2PEvent.State(conversationId, false))
        }
    }

    suspend fun stop() {
        for (id in sessions.keys.toList()) stopSession(id)
        scope.cancel()
    }

    private fun session(id: Long) = sessions[id] ?: throw NoSuchElementException("нет активной сессии для беседы $id")

    private suspend fun connect(id: Long, roomId: ByteArray, prs: ByteArray, timeoutMs: Long) {
        if (sessions.containsKey(id)) return // канал уже поднят
        var joined: Rendezvous? = null
        try {
            val rv = joinRendezvous(http, rendezvousUrl, roomId, timeoutMs).also { joined = it }
            val transport = RelayTransport(rv.ws)
            val result = withTimeoutOrNull(handshakeTimeoutMs) { handshake(transport, prs, rv.role, ristretto) }
                ?: throw PakeException("собеседник не завершил хендшейк")
            val session = openSession(
                store.session(id), transport, result.sk, result.ratchetState,
                onMessage = { mid -> _events.tryEmit(P2PEvent.Message(id, mid)) },
                onDisconnect = { onDisconnect(id) },
            )
            sessions[id] = session
            roles[id] = rv.role
            session.start(scope)
            if (rv.role == Role.INITIATOR) runCatching { session.sendPrime() } // best-effort
            _events.tryEmit(P2PEvent.State(id, true))
        } catch (e: Exception) {
            joined?.ws?.close()
            if (e !is CancellationException) _events.tryEmit(P2PEvent.Error(id, e))
            throw e
        }
    }

    private fun onDisconnect(id: Long) {
        if (sessions.remove(id) != null) {
            roles.remove(id)
            _events.tryEmit(P2PEvent.State(id, false))
        }
    }
}
```

```kotlin
package ru.mobile.mys.storage

import ru.mobile.mys.crypto.RatchetState
import ru.mobile.mys.p2p.P2PStore
import ru.mobile.mys.p2p.SessionStore

/** P2P-беседы в зашифрованном vault (режим "decentralized", как у десктопа). */
class VaultP2PStore(private val vault: Vault) : P2PStore {
    override fun conversationFor(roomId: ByteArray): Long =
        vault.conversations.getByRoomId(roomId, MODE)?.id ?: vault.conversations.add(MODE, roomId)

    override fun setPrs(conversationId: Long, prs: ByteArray) = vault.conversations.setPrs(conversationId, prs)

    override fun roomAndPrs(conversationId: Long): Pair<ByteArray, ByteArray>? {
        val c = vault.conversations.get(conversationId) ?: return null
        val room = c.roomId ?: return null
        val prs = c.p2pPrs ?: return null
        return room to prs
    }

    override fun session(conversationId: Long): SessionStore = object : SessionStore {
        override fun loadRatchet() = vault.ratchet.load(conversationId)
        override fun saveRatchet(state: RatchetState) = vault.ratchet.save(conversationId, state)
        override fun addOutgoing(body: ByteArray, status: String, kind: String, filename: String?, mimeType: String?) =
            vault.messages.add(conversationId, "out", body, status, kind, filename, mimeType)
        override fun setStatus(messageId: Long, status: String) = vault.messages.setStatus(messageId, status)
        override fun receive(body: ByteArray, state: RatchetState, kind: String, filename: String?, mimeType: String?) =
            vault.receiveMessage(conversationId, body, state, kind, filename, mimeType)
    }

    private companion object {
        const val MODE = "decentralized"
    }
}
```
