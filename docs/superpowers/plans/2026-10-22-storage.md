> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Зашифрованное локальное хранилище МЫС на Android: SQLCipher-vault под мастер-паролем (Argon2id), блокировка/duress как в десктопе, репозитории под P2P и обёртка ключа для биометрии.

**Tech Stack:** SQLCipher for Android 4.19.1, lazysodium-android 5.2.0 (`crypto_pwhash`), Android Keystore, androidx.test (runner, ext-junit), kotlin-test-junit.

Спека: `docs/superpowers/specs/2026-10-22-storage.md`. Все команды — из корня `vsc_mobile` внутри `nix develop`.

```kotlin
package ru.mobile.mys.storage

import java.io.File
import java.io.FileOutputStream
import java.util.Base64
import java.util.Properties

/**
 * Открытый sidecar рядом с БД (аналог vault.meta.json десктопа): несекретные
 * параметры KDF и счётчики неверных попыток. Секретов здесь нет по построению.
 */
class Sidecar(
    val salt: ByteArray,
    val opsLimit: Long = 3,
    val memLimit: Long = 128L shl 20,
    var failed: Int = 0,
    var lockoutUntil: Long = 0,
    var wipeEnabled: Boolean = false,
    var wipeThreshold: Int = 10,
) {
    fun lockRemaining(now: Long): Long = maxOf(0L, lockoutUntil - now)

    /** Учесть неверный пароль. true — порог duress достигнут, vault нужно стереть. */
    fun registerFailure(now: Long): Boolean {
        failed += 1
        if (wipeEnabled && failed >= wipeThreshold) return true
        lockoutUntil = now + delayMs(failed)
        return false
    }

    fun resetAttempts() {
        failed = 0
        lockoutUntil = 0
    }

    /** Атомарная запись: tmp + fsync + rename. */
    fun write(file: File) {
        val p = Properties()
        p["format_version"] = "1"
        p["salt"] = Base64.getEncoder().encodeToString(salt)
        p["ops_limit"] = opsLimit.toString()
        p["mem_limit"] = memLimit.toString()
        p["failed"] = failed.toString()
        p["lockout_until"] = lockoutUntil.toString()
        p["wipe_enabled"] = wipeEnabled.toString()
        p["wipe_threshold"] = wipeThreshold.toString()
        val tmp = File(file.path + ".tmp")
        FileOutputStream(tmp).use { out ->
            p.store(out, null)
            out.fd.sync()
        }
        check(tmp.renameTo(file)) { "не удалось записать sidecar" }
    }

    companion object {
        const val SALT_LEN = 16
        const val LOCK_CAP_MS = 300_000L

        /** 2^failed секунд, не больше 5 минут — как в десктопе. */
        fun delayMs(failed: Int): Long = minOf(1000L shl minOf(failed, 30), LOCK_CAP_MS)

        fun read(file: File): Sidecar {
            val p = Properties().apply { file.inputStream().use { load(it) } }
            return Sidecar(
                salt = Base64.getDecoder().decode(p.getProperty("salt")),
                opsLimit = p.getProperty("ops_limit").toLong(),
                memLimit = p.getProperty("mem_limit").toLong(),
                failed = p.getProperty("failed").toInt(),
                lockoutUntil = p.getProperty("lockout_until").toLong(),
                wipeEnabled = p.getProperty("wipe_enabled").toBoolean(),
                wipeThreshold = p.getProperty("wipe_threshold").toInt(),
            )
        }
    }
}
```

```kotlin
package ru.mobile.mys.storage

import android.content.ContentValues
import android.database.Cursor
import net.zetetic.database.sqlcipher.SQLiteDatabase

/*
 * Схема = итоговая схема десктопа (mys_storage/schema.py, main v7) одной миграцией:
 * БД телефона с десктопом не обменивается. Следующие изменения — новым элементом списка.
 */
internal val MIGRATIONS: List<List<String>> = listOf(
    listOf(
        "CREATE TABLE settings (key TEXT PRIMARY KEY, value TEXT NOT NULL)",
        "CREATE TABLE conversations (" +
            " id INTEGER PRIMARY KEY, mode TEXT NOT NULL, room_id BLOB, title TEXT," +
            " created_at INTEGER NOT NULL, p2p_prs BLOB)",
        "CREATE TABLE messages (" +
            " id INTEGER PRIMARY KEY," +
            " conversation_id INTEGER NOT NULL REFERENCES conversations(id)," +
            " direction TEXT NOT NULL, body BLOB, status TEXT NOT NULL," +
            " kind TEXT NOT NULL DEFAULT 'text', filename TEXT, mime_type TEXT," +
            " sender TEXT, wire_seq INTEGER, client_msg_id TEXT, media_ref TEXT," +
            " reply_author TEXT, reply_snippet TEXT, sent_at INTEGER, received_at INTEGER)",
        "CREATE INDEX idx_messages_conv ON messages(conversation_id, id)",
        "CREATE INDEX idx_messages_conv_wire ON messages(conversation_id, wire_seq)",
        "CREATE TABLE ratchet_state (" +
            " conversation_id INTEGER PRIMARY KEY REFERENCES conversations(id)," +
            " state_blob BLOB NOT NULL, updated_at INTEGER NOT NULL)",
    ),
)

internal fun migrate(db: SQLiteDatabase) {
    for (v in db.version until MIGRATIONS.size) db.tx {
        MIGRATIONS[v].forEach { db.execSQL(it) }
        db.version = v + 1
    }
}

internal inline fun <T> SQLiteDatabase.tx(block: () -> T): T {
    beginTransaction()
    try {
        return block().also { setTransactionSuccessful() }
    } finally {
        endTransaction()
    }
}

internal fun <T> SQLiteDatabase.select(sql: String, vararg args: Any?, row: (Cursor) -> T): List<T> =
    rawQuery(sql, arrayOf(*args)).use { c -> buildList { while (c.moveToNext()) add(row(c)) } }

internal fun SQLiteDatabase.insertRow(table: String, vararg values: Pair<String, Any?>): Long {
    val cv = ContentValues()
    for ((k, v) in values) when (v) {
        null -> cv.putNull(k)
        is ByteArray -> cv.put(k, v)
        is String -> cv.put(k, v)
        is Long -> cv.put(k, v)
        is Int -> cv.put(k, v)
        else -> error("неподдерживаемый тип ${v::class}")
    }
    return insert(table, SQLiteDatabase.CONFLICT_ABORT, cv).also { check(it != -1L) { "insert в $table" } }
}

internal fun Cursor.blobOrNull(i: Int): ByteArray? = if (isNull(i)) null else getBlob(i)
internal fun Cursor.stringOrNull(i: Int): String? = if (isNull(i)) null else getString(i)
internal fun Cursor.longOrNull(i: Int): Long? = if (isNull(i)) null else getLong(i)
```

```kotlin
package ru.mobile.mys.storage

import android.database.Cursor
import net.zetetic.database.sqlcipher.SQLiteDatabase
import ru.mobile.mys.crypto.RatchetState
import ru.mobile.mys.crypto.deserializeState
import ru.mobile.mys.crypto.serializeState

class Conversation(
    val id: Long, val mode: String, val roomId: ByteArray?, val title: String?,
    val createdAt: Long, val p2pPrs: ByteArray?,
)

class Message(
    val id: Long, val conversationId: Long, val direction: String, val body: ByteArray,
    val status: String, val kind: String, val filename: String?, val mimeType: String?,
    val sender: String?, val wireSeq: Long?, val clientMsgId: String?,
    val sentAt: Long?, val receivedAt: Long?,
)

class SettingsRepo internal constructor(private val db: SQLiteDatabase) {
    fun get(key: String): String? = db.select("SELECT value FROM settings WHERE key=?", key) { it.getString(0) }.firstOrNull()

    fun set(key: String, value: String) = db.execSQL(
        "INSERT INTO settings(key, value) VALUES(?, ?) ON CONFLICT(key) DO UPDATE SET value=excluded.value",
        arrayOf(key, value),
    )

    fun delete(key: String) = db.execSQL("DELETE FROM settings WHERE key=?", arrayOf(key))

    fun deletePrefix(prefix: String) =
        db.execSQL("DELETE FROM settings WHERE substr(key, 1, ?)=?", arrayOf(prefix.length, prefix))
}

class ConversationsRepo internal constructor(private val db: SQLiteDatabase) {
    private val cols = "SELECT id, mode, room_id, title, created_at, p2p_prs FROM conversations"
    private fun row(c: Cursor) = Conversation(c.getLong(0), c.getString(1), c.blobOrNull(2), c.stringOrNull(3), c.getLong(4), c.blobOrNull(5))

    fun add(mode: String, roomId: ByteArray? = null, title: String? = null): Long = db.insertRow(
        "conversations", "mode" to mode, "room_id" to roomId, "title" to title,
        "created_at" to System.currentTimeMillis(),
    )

    fun get(id: Long): Conversation? = db.select("$cols WHERE id=?", id) { row(it) }.firstOrNull()

    fun getByRoomId(roomId: ByteArray, mode: String): Conversation? =
        db.select("$cols WHERE room_id=? AND mode=?", roomId, mode) { row(it) }.firstOrNull()

    fun list(mode: String? = null): List<Conversation> =
        if (mode == null) db.select("$cols ORDER BY id") { row(it) }
        else db.select("$cols WHERE mode=? ORDER BY id", mode) { row(it) }

    fun setPrs(id: Long, prs: ByteArray) = db.execSQL("UPDATE conversations SET p2p_prs=? WHERE id=?", arrayOf(prs, id))

    fun rename(id: Long, title: String) = db.execSQL("UPDATE conversations SET title=? WHERE id=?", arrayOf(title, id))

    /** Беседа вместе с сообщениями и состоянием ratchet. */
    fun delete(id: Long) = db.tx {
        db.execSQL("DELETE FROM messages WHERE conversation_id=?", arrayOf(id))
        db.execSQL("DELETE FROM ratchet_state WHERE conversation_id=?", arrayOf(id))
        db.execSQL("DELETE FROM conversations WHERE id=?", arrayOf(id))
    }
}

class MessagesRepo internal constructor(private val db: SQLiteDatabase) {
    private val cols = "SELECT id, conversation_id, direction, body, status, kind, filename, mime_type," +
        " sender, wire_seq, client_msg_id, sent_at, received_at FROM messages"

    private fun row(c: Cursor) = Message(
        c.getLong(0), c.getLong(1), c.getString(2), c.blobOrNull(3) ?: ByteArray(0), c.getString(4),
        c.getString(5), c.stringOrNull(6), c.stringOrNull(7), c.stringOrNull(8), c.longOrNull(9),
        c.stringOrNull(10), c.longOrNull(11), c.longOrNull(12),
    )

    fun add(
        conversationId: Long, direction: String, body: ByteArray, status: String,
        kind: String = "text", filename: String? = null, mimeType: String? = null,
        sender: String? = null, wireSeq: Long? = null, clientMsgId: String? = null,
    ): Long {
        val now = System.currentTimeMillis()
        return db.insertRow(
            "messages", "conversation_id" to conversationId, "direction" to direction, "body" to body,
            "status" to status, "kind" to kind, "filename" to filename, "mime_type" to mimeType,
            "sender" to sender, "wire_seq" to wireSeq, "client_msg_id" to clientMsgId,
            (if (direction == "out") "sent_at" else "received_at") to now,
        )
    }

    fun get(id: Long): Message? = db.select("$cols WHERE id=?", id) { row(it) }.firstOrNull()

    fun list(conversationId: Long): List<Message> = db.select("$cols WHERE conversation_id=? ORDER BY id", conversationId) { row(it) }

    fun setStatus(id: Long, status: String) = db.execSQL("UPDATE messages SET status=? WHERE id=?", arrayOf(status, id))

    fun deleteForConversation(conversationId: Long) = db.execSQL("DELETE FROM messages WHERE conversation_id=?", arrayOf(conversationId))
}

class RatchetRepo internal constructor(private val db: SQLiteDatabase) {
    fun save(conversationId: Long, state: RatchetState) = db.execSQL(
        "INSERT INTO ratchet_state(conversation_id, state_blob, updated_at) VALUES(?, ?, ?)" +
            " ON CONFLICT(conversation_id) DO UPDATE SET state_blob=excluded.state_blob, updated_at=excluded.updated_at",
        arrayOf(conversationId, serializeState(state), System.currentTimeMillis()),
    )

    fun load(conversationId: Long): RatchetState? =
        db.select("SELECT state_blob FROM ratchet_state WHERE conversation_id=?", conversationId) { deserializeState(it.getBlob(0)) }
            .firstOrNull()

    fun delete(conversationId: Long) = db.execSQL("DELETE FROM ratchet_state WHERE conversation_id=?", arrayOf(conversationId))
}
```

```kotlin
package ru.mobile.mys.storage

import java.io.Closeable
import java.io.File
import java.io.RandomAccessFile
import java.security.MessageDigest
import java.security.SecureRandom
import net.zetetic.database.sqlcipher.SQLiteDatabase
import ru.mobile.mys.crypto.RatchetState

sealed class VaultException(message: String) : Exception(message)
class VaultExists : VaultException("vault уже существует")
class WrongPassword : VaultException("неверный пароль")
class VaultLocked(val remainingMs: Long) : VaultException("vault заблокирован ещё на ${remainingMs / 1000} с")

/**
 * Зашифрованный vault (SQLCipher) в отдельном каталоге `dir`: `vault.db` + открытый
 * `vault.meta` (+ `vault.bio`, если включена биометрия). Порт mys_storage/vault.py.
 */
class Vault private constructor(private val db: SQLiteDatabase, private val dir: File, key: ByteArray) : Closeable {
    /** Сырой ключ БД — нужен только для обёртки биометрией (BiometricKey.seal). */
    internal var key: ByteArray = key
        private set

    val settings = SettingsRepo(db)
    val conversations = ConversationsRepo(db)
    val messages = MessagesRepo(db)
    val ratchet = RatchetRepo(db)

    /** Входящее сообщение и новое состояние ratchet — одной транзакцией. */
    fun receiveMessage(
        conversationId: Long, body: ByteArray, newState: RatchetState,
        kind: String = "text", filename: String? = null, mimeType: String? = null, wireSeq: Long? = null,
    ): Long = db.tx {
        messages.add(conversationId, "in", body, "received", kind, filename, mimeType, wireSeq = wireSeq)
            .also { ratchet.save(conversationId, newState) }
    }

    fun changePassword(old: ByteArray, new: ByteArray) {
        val sc = Sidecar.read(metaFile(dir))
        val check = deriveDbKey(old, sc)
        val same = MessageDigest.isEqual(check, key)
        check.fill(0)
        if (!same) throw WrongPassword()
        val fresh = Sidecar(randomSalt(), sc.opsLimit, sc.memLimit, wipeEnabled = sc.wipeEnabled, wipeThreshold = sc.wipeThreshold)
        val newKey = deriveDbKey(new, fresh)
        db.changePassword(rawKey(newKey))
        fresh.write(metaFile(dir))
        key.fill(0)
        key = newKey
    }

    override fun close() {
        db.close()
        key.fill(0)
    }

    companion object {
        init { System.loadLibrary("sqlcipher") }

        private fun dbFile(dir: File) = File(dir, "vault.db")
        private fun metaFile(dir: File) = File(dir, "vault.meta")
        private fun randomSalt() = ByteArray(Sidecar.SALT_LEN).also { SecureRandom().nextBytes(it) }

        /** SQLCipher raw key: `x'<hex>'` — без внутреннего PBKDF2, ключ уже из Argon2id. */
        private fun rawKey(key: ByteArray) = "x'${key.joinToString("") { "%02x".format(it) }}'".toByteArray()

        fun exists(dir: File) = dbFile(dir).exists() || metaFile(dir).exists()

        fun create(dir: File, password: ByteArray, opsLimit: Long = 3, memLimit: Long = 128L shl 20): Vault {
            if (exists(dir)) throw VaultExists()
            dir.mkdirs()
            val sc = Sidecar(randomSalt(), opsLimit, memLimit)
            val key = deriveDbKey(password, sc)
            val raw = rawKey(key)
            val db = SQLiteDatabase.openOrCreateDatabase(dbFile(dir), raw, null, null)
            raw.fill(0)
            migrate(db)
            sc.write(metaFile(dir))
            return Vault(db, dir, key)
        }

        fun open(dir: File, password: ByteArray, now: Long = System.currentTimeMillis()): Vault {
            val meta = metaFile(dir)
            val sc = Sidecar.read(meta)
            sc.lockRemaining(now).let { if (it > 0) throw VaultLocked(it) }
            val key = deriveDbKey(password, sc)
            val db = tryOpen(dir, key)
            if (db == null) {
                key.fill(0)
                if (sc.registerFailure(now)) wipe(dir) else sc.write(meta)
                throw WrongPassword()
            }
            sc.resetAttempts()
            sc.write(meta)
            migrate(db)
            return Vault(db, dir, key)
        }

        /** Разблокировка готовым ключом (из обёртки биометрии). */
        fun openWithKey(dir: File, key: ByteArray): Vault {
            val db = tryOpen(dir, key) ?: throw WrongPassword()
            migrate(db)
            return Vault(db, dir, key.copyOf())
        }

        /** Стереть весь каталог vault (duress или сброс). */
        fun wipe(dir: File) {
            // ponytail: перезапись нулями на флеше не гарантирует стирания (wear leveling);
            // защиту даёт шифрование, перезапись — best-effort, как в десктопе.
            dir.listFiles()?.filter { it.isFile }?.forEach { f ->
                runCatching {
                    RandomAccessFile(f, "rw").use { raf ->
                        val zeros = ByteArray(64 * 1024)
                        var left = raf.length()
                        while (left > 0) {
                            val n = minOf(left, zeros.size.toLong()).toInt()
                            raf.write(zeros, 0, n)
                            left -= n
                        }
                        raf.fd.sync()
                    }
                }
                f.delete()
            }
        }

        /** Открыть существующую БД ключом; null — ключ не подошёл. Не создаёт файл. */
        private fun tryOpen(dir: File, key: ByteArray): SQLiteDatabase? {
            check(dbFile(dir).exists()) { "нет файла vault.db" }
            val raw = rawKey(key)
            return try {
                SQLiteDatabase.openDatabase(dbFile(dir).path, raw, null, SQLiteDatabase.OPEN_READWRITE, null, null).also { db ->
                    try {
                        db.select("SELECT count(*) FROM sqlite_master") { it.getLong(0) }
                    } catch (e: Exception) {
                        db.close()
                        throw e
                    }
                }
            } catch (e: Exception) {
                null
            } finally {
                raw.fill(0)
            }
        }
    }
}
```

```kotlin
package ru.mobile.mys.storage

import android.os.Build
import android.security.keystore.KeyGenParameterSpec
import android.security.keystore.KeyPermanentlyInvalidatedException
import android.security.keystore.KeyProperties
import java.io.File
import java.security.KeyStore
import java.security.ProviderException
import javax.crypto.Cipher
import javax.crypto.KeyGenerator
import javax.crypto.SecretKey
import javax.crypto.spec.GCMParameterSpec

/**
 * Обёртка ключа БД AES-GCM ключом Android Keystore для разблокировки биометрией.
 *
 * Шифр из [encryptCipher]/[decryptCipher] передаётся в `BiometricPrompt.CryptoObject`;
 * после успешной аутентификации — в [seal]/[open]. Файл `vault.bio` = iv(12) ‖ ct.
 * `requireAuth = false` — только для тестов (ключ без биометрии).
 */
class BiometricKey(private val dir: File, private val requireAuth: Boolean = true) {
    private val file get() = File(dir, "vault.bio")

    fun isEnabled() = file.exists()

    /** Шифр для включения биометрии (создаёт новый ключ Keystore). */
    fun encryptCipher(): Cipher {
        createKey()
        return Cipher.getInstance(TRANSFORM).apply { init(Cipher.ENCRYPT_MODE, key()) }
    }

    fun seal(cipher: Cipher, dbKey: ByteArray) = file.writeBytes(cipher.iv + cipher.doFinal(dbKey))

    /** null — обёртки нет или ключ инвалидирован (сменились отпечатки): только пароль. */
    fun decryptCipher(): Cipher? {
        if (!file.exists()) return null
        val key = key() ?: run { disable(); return null }
        return try {
            Cipher.getInstance(TRANSFORM).apply { init(Cipher.DECRYPT_MODE, key, GCMParameterSpec(128, file.readBytes(), 0, IV_LEN)) }
        } catch (e: KeyPermanentlyInvalidatedException) {
            disable()
            null
        }
    }

    fun open(cipher: Cipher): ByteArray {
        val blob = file.readBytes()
        return cipher.doFinal(blob, IV_LEN, blob.size - IV_LEN)
    }

    fun disable() {
        file.delete()
        keyStore().deleteEntry(ALIAS)
    }

    private fun keyStore() = KeyStore.getInstance("AndroidKeyStore").apply { load(null) }

    private fun key() = keyStore().getKey(ALIAS, null) as SecretKey?

    private fun createKey() {
        fun spec(strongBox: Boolean) = KeyGenParameterSpec.Builder(ALIAS, KeyProperties.PURPOSE_ENCRYPT or KeyProperties.PURPOSE_DECRYPT)
            .setBlockModes(KeyProperties.BLOCK_MODE_GCM)
            .setEncryptionPaddings(KeyProperties.ENCRYPTION_PADDING_NONE)
            .setKeySize(256)
            .apply {
                if (requireAuth) {
                    setUserAuthenticationRequired(true)
                    setInvalidatedByBiometricEnrollment(true)
                    if (Build.VERSION.SDK_INT >= 30) setUserAuthenticationParameters(0, KeyProperties.AUTH_BIOMETRIC_STRONG)
                }
                if (strongBox && Build.VERSION.SDK_INT >= 28) setIsStrongBoxBacked(true)
            }
            .build()
        val gen = KeyGenerator.getInstance(KeyProperties.KEY_ALGORITHM_AES, "AndroidKeyStore")
        try {
            gen.init(spec(strongBox = true))
            gen.generateKey()
        } catch (e: ProviderException) { // StrongBoxUnavailableException (API 28+) — её предок
            gen.init(spec(strongBox = false))
            gen.generateKey()
        }
    }

    private companion object {
        const val ALIAS = "mys.vault.bio"
        const val TRANSFORM = "AES/GCM/NoPadding"
        const val IV_LEN = 12
    }
}
```
