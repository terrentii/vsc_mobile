> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Централизованный режим МЫС на Kotlin: аккаунт, комнаты (создание, вход по коду), сообщения с live-доставкой, ответы, правка/удаление, медиа — совместимо с `vsc_web` и поведением десктопа.

**Architecture:** JVM-модуль `:central` (RestClient + WsClient на OkHttp, SyncEngine и CentralService на корутинах) поверх интерфейса `CentralStore`; `VaultCentralStore` в `:app` на зашифрованном vault. Проверка — живые JVM-тесты против настоящего `vsc_web` и инструментальный тест адаптера.

**Tech Stack:** Kotlin 2.1.21, kotlinx-coroutines 1.10.2, OkHttp 4.12.0, org.json (платформенный на Android).

package ru.mobile.mys.central

import java.time.LocalDateTime
import java.time.OffsetDateTime
import java.time.ZoneOffset
import org.json.JSONObject

open class CentralException(message: String, cause: Throwable? = null) : Exception(message, cause)

/** Неверные учётные данные или протухший токен (HTTP 401) — терминально. */
class AuthException(message: String = "unauthorized") : CentralException(message)

/** 4xx кроме 401; [code] — код ошибки сервера: not_found, forbidden, rate_limited… */
class ServerException(val code: String) : CentralException(code)

/** Сеть недоступна, обрыв, 5xx. */
class NetworkException(message: String, cause: Throwable? = null) : CentralException(message, cause)

/** Ответ сервера не по контракту. */
class ProtocolException(message: String, cause: Throwable? = null) : CentralException(message, cause)

class Session(val serverUrl: String, val username: String, val userId: Long, val token: String)

class Room(val id: Long, val name: String?, val isDirect: Boolean = false, val updatedAt: String? = null, val code: String? = null)

/** Готовая цитата от сервера: {id, sender, body ≤ 60}. */
class Reply(val id: Long, val sender: String, val body: String)

class RemoteMessage(
    val id: Long, val roomId: Long, val sender: String, val body: String, val createdAt: String,
    val clientMsgId: String? = null, val media: String? = null, val reply: Reply? = null,
)

class UploadedMedia(val filename: String, val mimeType: String?, val size: Long?)

internal fun JSONObject.str(key: String): String? = if (!has(key) || isNull(key)) null else getString(key)

fun parseRoom(o: JSONObject) = Room(o.getLong("id"), o.str("name"), o.optBoolean("is_direct", false), o.str("updated_at"), o.str("code"))

fun parseMessage(o: JSONObject, roomId: Long) = RemoteMessage(
    id = o.getLong("id"),
    roomId = if (o.has("room_id") && !o.isNull("room_id")) o.getLong("room_id") else roomId,
    sender = o.getString("sender"),
    body = o.str("body") ?: "",
    createdAt = o.getString("created_at"),
    clientMsgId = o.str("client_msg_id"),
    media = o.str("media"),
    reply = o.optJSONObject("reply_to")?.let { Reply(it.getLong("id"), it.str("sender") ?: "", it.str("body") ?: "") },
)

/** `https://host` → `wss://host/ws`; нераспознанная схема — null (без live-канала). */
fun wsUrlFor(serverUrl: String): String? {
    for ((http, ws) in listOf("https://" to "wss://", "http://" to "ws://")) {
        if (serverUrl.startsWith(http)) return ws + serverUrl.removePrefix(http).trimEnd('/') + "/ws"
    }
    return null
}

/** Серверный ISO `created_at` (наивный = UTC) → epoch ms; непарсибельное — null. */
fun epochMs(createdAt: String): Long? =
    runCatching { OffsetDateTime.parse(createdAt).toInstant().toEpochMilli() }.getOrNull()
        ?: runCatching { LocalDateTime.parse(createdAt).toInstant(ZoneOffset.UTC).toEpochMilli() }.getOrNull()

const val MAX_MEDIA_SIZE = 50 * 1024 * 1024 // защита памяти телефона; сервер лимитов не держит

private val IMAGE_EXTS = setOf("png", "jpg", "jpeg", "gif", "webp", "bmp", "tiff", "tif")

fun kindForFilename(name: String) = if (name.substringAfterLast('.', "").lowercase() in IMAGE_EXTS) "image" else "file"

/** Серверное имя `<uuid32hex>_<original>` → отображаемое (паритет с вебом). */
fun displayName(mediaRef: String) = mediaRef.substringAfter('_', "").ifEmpty { mediaRef }

package ru.mobile.mys.central

/** CentralStore в памяти — для живых тестов. */
class MemoryCentralStore : CentralStore {
    class Row(val id: Long, val m: NewMessage) {
        var body: ByteArray? = m.body
        var status: String = m.status
        var wireSeq: Long? = m.wireSeq
        fun stored() = StoredMessage(id, m.conversationId, m.direction, body, status, wireSeq, m.mediaRef, m.kind, m.sender)
    }

    private val settings = HashMap<String, String>()
    private val convs = LinkedHashMap<Long, Pair<Long, String?>>() // conv -> (room, title)
    val rows = mutableListOf<Row>()
    private var next = 1L

    @Synchronized override fun setting(key: String) = settings[key]
    @Synchronized override fun setSetting(key: String, value: String) { settings[key] = value }
    @Synchronized override fun deleteSetting(key: String) { settings.remove(key) }
    @Synchronized override fun deleteSettings(prefix: String) { settings.keys.removeAll { it.startsWith(prefix) } }

    @Synchronized override fun conversationForRoom(roomId: Long, title: String?, create: Boolean): Long? =
        convs.entries.firstOrNull { it.value.first == roomId }?.key
            ?: if (create) next++.also { convs[it] = roomId to title } else null

    @Synchronized override fun roomOf(conversationId: Long) = convs.getValue(conversationId).first
    @Synchronized fun conversations() = convs.keys.toList()

    @Synchronized override fun wipeConversations() { rows.clear(); convs.clear() }

    @Synchronized override fun add(m: NewMessage): Long = next++.also { rows += Row(it, m) }
    @Synchronized override fun get(messageId: Long) = rows.find { it.id == messageId }?.stored()
    @Synchronized fun messages(conv: Long) = rows.filter { it.m.conversationId == conv }.map { it.stored() }

    @Synchronized override fun findByWire(conversationId: Long, wireSeq: Long) =
        rows.find { it.m.conversationId == conversationId && it.wireSeq == wireSeq }?.stored()

    @Synchronized override fun findOutByClientId(conversationId: Long, clientMsgId: String) =
        rows.find { it.m.conversationId == conversationId && it.m.direction == "out" && it.m.clientMsgId == clientMsgId }?.stored()

    @Synchronized override fun findUnconfirmedOutByBody(conversationId: Long, body: ByteArray) =
        rows.find { it.m.conversationId == conversationId && it.m.direction == "out" && it.wireSeq == null && it.body?.contentEquals(body) == true }?.stored()

    @Synchronized override fun markSent(messageId: Long, wireSeq: Long) { rows.first { it.id == messageId }.apply { this.wireSeq = wireSeq; status = "sent" } }
    @Synchronized override fun setStatus(messageId: Long, status: String) { rows.first { it.id == messageId }.status = status }
    @Synchronized override fun setBody(messageId: Long, body: ByteArray) { rows.first { it.id == messageId }.body = body }
    @Synchronized override fun delete(messageId: Long) { rows.removeAll { it.id == messageId } }
}

package ru.mobile.mys.central

/** Сессия и настройки «Центра» в зашифрованном хранилище — ключи как у десктопа. */
internal object Account {
    const val K_SERVER = "central.server_url"
    const val K_USER = "central.username"
    const val K_UID = "central.user_id"
    const val K_TOKEN = "central.token"
    const val K_WIPE = "central.wipe_on_logout"
    const val CURSOR_PREFIX = "central.cursor."
    const val CODE_PREFIX = "central.code."

    fun save(store: CentralStore, s: Session) {
        store.setSetting(K_SERVER, s.serverUrl)
        store.setSetting(K_USER, s.username)
        store.setSetting(K_UID, s.userId.toString())
        store.setSetting(K_TOKEN, s.token)
    }

    fun load(store: CentralStore): Session? {
        val token = store.setting(K_TOKEN) ?: return null
        return Session(store.setting(K_SERVER) ?: return null, store.setting(K_USER) ?: return null,
            store.setting(K_UID)?.toLongOrNull() ?: return null, token)
    }

    fun clear(store: CentralStore) = listOf(K_SERVER, K_USER, K_UID, K_TOKEN).forEach(store::deleteSetting)

    /** Беседы, сообщения, курсоры и коды «Центра»; сессию чистит [clear]. */
    fun wipeCache(store: CentralStore) {
        store.wipeConversations()
        store.deleteSettings(CURSOR_PREFIX)
        store.deleteSettings(CODE_PREFIX)
    }
}

package ru.mobile.mys.central

import java.io.IOException
import kotlinx.coroutines.Dispatchers

/** HTTP-контракт сервера и маппинг статусов в ошибки. О хранилище не знает. */
class RestClient(baseUrl: String, var token: String? = null, private val http: OkHttpClient = OkHttpClient()) {
    val baseUrl = baseUrl.trimEnd('/')
    private val jsonType = "application/json; charset=utf-8".toMediaType()

    suspend fun register(username: String, password: String) = session(post("/api/auth/register", credentials(username, password), auth = false))
    suspend fun login(username: String, password: String) = session(post("/api/auth/login", credentials(username, password), auth = false))

    suspend fun logout() {
        try { post("/api/auth/logout", JSONObject()) } finally { token = null }
    }

    suspend fun listRooms(): List<Room> {
        val o = get("/api/rooms")
        return parse("rooms") {
            val arr = o.getJSONArray("rooms")
            List(arr.length()) { parseRoom(arr.getJSONObject(it)) }
        }
    }

    suspend fun createRoom(name: String): Room = post("/api/rooms", JSONObject().put("name", name)).let { parse("room") { parseRoom(it) } }

    /** Войти по 10-значному коду. ServerException: bad_request / not_found / forbidden. */
    suspend fun joinRoom(code: String): Room = post("/api/rooms/join", JSONObject().put("code", code)).let { parse("room") { parseRoom(it) } }

    /** (сообщения, next_cursor). */
    suspend fun getMessages(roomId: Long, after: Long? = null, limit: Int? = null): Pair<List<RemoteMessage>, Long?> {
        val u = "$baseUrl/api/rooms/$roomId/messages".toHttpUrl().newBuilder().apply {
            after?.let { addQueryParameter("after", it.toString()) }
            limit?.let { addQueryParameter("limit", it.toString()) }
        }.build()
        val o = json(call(Request.Builder().url(u)))
        return parse("messages") {
            val arr = o.getJSONArray("messages")
            List(arr.length()) { parseMessage(arr.getJSONObject(it), roomId) } to
                (if (o.has("next_cursor") && !o.isNull("next_cursor")) o.getLong("next_cursor") else null)
        }
    }

    suspend fun postMessage(roomId: Long, body: String, clientMsgId: String, media: String? = null, replyTo: Long? = null): RemoteMessage {
        val p = JSONObject().put("room_id", roomId).put("body", body).put("client_msg_id", clientMsgId)
        media?.let { p.put("media", it) }
        replyTo?.let { p.put("reply_to", it) }
        return post("/api/messages", p).let { parse("message") { parseMessage(it, roomId) } }
    }

    suspend fun editMessage(messageId: Long, body: String) { post("/api/messages/$messageId/edit", JSONObject().put("body", body)) }

    suspend fun deleteMessage(messageId: Long) { post("/api/messages/$messageId/delete", JSONObject()) }

    suspend fun uploadMedia(roomId: Long, filename: String, data: ByteArray, mimeType: String): UploadedMedia {
        val form = MultipartBody.Builder().setType(MultipartBody.FORM)
            .addFormDataPart("file", filename, data.toRequestBody(mimeType.toMediaTypeOrNull())).build()
        val o = json(call(Request.Builder().url("$baseUrl/api/rooms/$roomId/media").post(form)))
        return parse("upload") { UploadedMedia(o.getString("filename"), o.str("mime_type"), if (o.has("size")) o.getLong("size") else null) }
    }

    suspend fun downloadMedia(roomId: Long, filename: String): ByteArray =
        call(Request.Builder().url("$baseUrl/api/rooms/$roomId/media/$filename"))

    private fun credentials(u: String, p: String) = JSONObject().put("username", u).put("password", p)

    private fun session(o: JSONObject): Session = parse("auth") {
        val user = o.getJSONObject("user")
        Session(baseUrl, user.getString("username"), user.getLong("id"), o.getString("token")).also { token = it.token }
    }

    private suspend fun get(path: String) = json(call(Request.Builder().url(baseUrl + path)))

    private suspend fun post(path: String, body: JSONObject, auth: Boolean = true) =
        json(call(Request.Builder().url(baseUrl + path).post(body.toString().toRequestBody(jsonType)), auth))

    // ponytail: блокирующий execute на Dispatchers.IO — отмена корутины не рвёт запрос;
    // при нужде — okhttp Call.enqueue + suspendCancellableCoroutine.
    private suspend fun call(b: Request.Builder, auth: Boolean = true): ByteArray = withContext(Dispatchers.IO) {
        if (auth) token?.let { b.header("Authorization", "Bearer $it") }
        val resp = try { http.newCall(b.build()).execute() } catch (e: IOException) {
            throw NetworkException(e.message ?: "сеть недоступна", e)
        }
        resp.use { r ->
            val bytes = r.body?.bytes() ?: ByteArray(0)
            when {
                r.code == 401 -> throw AuthException()
                r.code >= 500 -> throw NetworkException("server status ${r.code}")
                r.code >= 400 -> throw ServerException(runCatching { JSONObject(bytes.decodeToString()).str("error") }.getOrNull() ?: "http ${r.code}")
            }
            bytes
        }
    }

    private fun json(b: ByteArray): JSONObject =
        try { JSONObject(b.decodeToString()) } catch (e: JSONException) { throw ProtocolException("invalid json in response", e) }

    private inline fun <T> parse(what: String, block: () -> T): T =
        try { block() } catch (e: JSONException) { throw ProtocolException("malformed $what response", e) }
}

package ru.mobile.mys.central

import kotlin.random.Random
import kotlinx.coroutines.CancellationException
import kotlinx.coroutines.channels.awaitClose
import kotlinx.coroutines.delay
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.callbackFlow
import kotlinx.coroutines.flow.flow
import okhttp3.OkHttpClient
import okhttp3.Request
import okhttp3.Response
import okhttp3.WebSocket
import okhttp3.WebSocketListener
import org.json.JSONException
import org.json.JSONObject

/** /ws: auth → ready → кадры сервера. Реконнект с backoff+джиттером; AuthException терминальна. */
class WsClient(
    private val url: String,
    private val token: String,
    private val http: OkHttpClient,
    private val initialBackoffMs: Long = 500,
    private val maxBackoffMs: Long = 30_000,
) {
    /** `{"type":"ready"}` при каждом (пере)подключении, затем кадры message/message_edited/message_deleted. */
    fun events(): Flow<JSONObject> = flow {
        var backoff = initialBackoffMs
        while (true) {
            try {
                session().collect { backoff = initialBackoffMs; emit(it) }
            } catch (e: AuthException) {
                throw e
            } catch (e: CancellationException) {
                throw e
            } catch (e: Exception) {
                // обрыв — переподключимся после паузы
            }
            delay(backoff + Random.nextLong(0, 300))
            backoff = minOf(backoff * 2, maxBackoffMs)
        }
    }

    private fun session(): Flow<JSONObject> = callbackFlow {
        val ws = http.newWebSocket(Request.Builder().url(url).build(), object : WebSocketListener() {
            var ready = false

            override fun onOpen(webSocket: WebSocket, response: Response) {
                webSocket.send(JSONObject().put("type", "auth").put("token", token).toString())
            }

            override fun onMessage(webSocket: WebSocket, text: String) {
                val o = try { JSONObject(text) } catch (e: JSONException) { close(ProtocolException("invalid ws frame")); return }
                if (!ready) {
                    when (o.optString("type")) {
                        "ready" -> ready = true
                        "error" -> { close(AuthException(o.optString("code", "unauthorized"))); return }
                        else -> { close(ProtocolException("unexpected first frame")); return }
                    }
                }
                trySend(o)
            }

            override fun onClosing(webSocket: WebSocket, code: Int, reason: String) {
                webSocket.close(1000, null)
                close()
            }

            override fun onClosed(webSocket: WebSocket, code: Int, reason: String) { close() }

            override fun onFailure(webSocket: WebSocket, t: Throwable, response: Response?) {
                close(NetworkException("ws: ${t.message}", t))
            }
        })
        awaitClose { ws.cancel() }
    }
}

package ru.mobile.mys.central

import java.net.URLConnection
import java.util.UUID
import org.json.JSONObject

/** Ссылка на сообщение, на которое отвечаем: серверный id + готовая цитата. */
class ReplyRef(val wire: Long, val sender: String, val snippet: String)

/**
 * Синк «Центра» — порт mys_centralized/sync.py: комнаты ↔ беседы, история по
 * курсору, дедуп (серверный id → client_msg_id → тело собственного эха),
 * оптимистичная отправка, правка/удаление, медиа с ленивой докачкой.
 * Курсор двигает только [syncHistory]; live-кадры лишь персистятся.
 */
class SyncEngine(
    private val store: CentralStore,
    private val rest: RestClient,
    private val onMessage: (conversationId: Long, messageId: Long) -> Unit = { _, _ -> },
    private val pageLimit: Int = 200,
    private val ownUsername: String? = null,
) {
    suspend fun syncRooms(): Map<Long, Long> = rest.listRooms().associate { it.id to remember(it) }

    suspend fun createRoom(name: String): Long = remember(rest.createRoom(name))

    suspend fun joinRoom(code: String): Long = remember(rest.joinRoom(code))

    private fun remember(room: Room): Long {
        room.code?.let { store.setSetting(Account.CODE_PREFIX + room.id, it) }
        return store.conversationForRoom(room.id, room.name)!!
    }

    suspend fun syncHistory(roomId: Long): Long {
        val conv = store.conversationForRoom(roomId, null)!!
        var cursor = store.setting(Account.CURSOR_PREFIX + roomId)?.toLongOrNull()
        while (true) {
            val (msgs, next) = rest.getMessages(roomId, cursor, pageLimit)
            if (msgs.isEmpty()) break
            var high = cursor ?: 0
            for (m in msgs) {
                ingest(conv, m)
                high = maxOf(high, m.id)
            }
            if (high > 0) store.setSetting(Account.CURSOR_PREFIX + roomId, high.toString())
            cursor = next ?: break
        }
        return conv
    }

    suspend fun syncAll(): Map<Long, Long> = syncRooms().also { rooms -> rooms.keys.forEach { syncHistory(it) } }

    fun ingestWs(frame: JSONObject): Long? {
        val roomId = frame.getLong("room_id")
        return ingest(store.conversationForRoom(roomId, null)!!, parseMessage(frame, roomId))
    }

    /** Live-правка: тело по серверному id. → conv или null. */
    fun applyWsEdit(frame: JSONObject): Long? {
        val conv = store.conversationForRoom(frame.getLong("room_id"), null, create = false) ?: return null
        val row = store.findByWire(conv, frame.getLong("id")) ?: return null
        store.setBody(row.id, (frame.str("body") ?: "").toByteArray())
        return conv
    }

    fun applyWsDelete(frame: JSONObject): Long? {
        val conv = store.conversationForRoom(frame.getLong("room_id"), null, create = false) ?: return null
        val row = store.findByWire(conv, frame.getLong("id")) ?: return null
        store.delete(row.id)
        return conv
    }

    /** pending → POST (идемпотентно по client_msg_id) → sent / failed; UI оповещается на каждом шаге. */
    suspend fun send(conv: Long, body: String, reply: ReplyRef? = null): Long {
        val roomId = store.roomOf(conv)
        val cid = newClientId()
        val id = store.add(NewMessage(conv, "out", body.toByteArray(), "pending", clientMsgId = cid,
            replyAuthor = reply?.sender, replySnippet = reply?.snippet))
        onMessage(conv, id)
        val msg = try {
            rest.postMessage(roomId, body, cid, replyTo = reply?.wire)
        } catch (e: Exception) {
            store.setStatus(id, "failed")
            onMessage(conv, id)
            throw e
        }
        store.markSent(id, msg.id)
        onMessage(conv, id)
        return id
    }

    suspend fun edit(messageId: Long, body: String) {
        val row = store.get(messageId) ?: throw IllegalArgumentException("сообщение не найдено")
        rest.editMessage(row.wireSeq ?: throw IllegalStateException("сообщение ещё не подтверждено сервером"), body)
        store.setBody(messageId, body.toByteArray())
    }

    suspend fun delete(messageId: Long) {
        val row = store.get(messageId) ?: return
        rest.deleteMessage(row.wireSeq ?: throw IllegalStateException("сообщение ещё не подтверждено сервером"))
        store.delete(messageId)
    }

    /** Загрузить файл, затем сообщение со ссылкой; байты уже в памяти — сохраняем сразу. */
    suspend fun sendFile(conv: Long, filename: String, mimeType: String, data: ByteArray): Long {
        require(data.size <= MAX_MEDIA_SIZE) { "файл больше ${MAX_MEDIA_SIZE / (1024 * 1024)} МБ" }
        val roomId = store.roomOf(conv)
        val up = rest.uploadMedia(roomId, filename, data, mimeType)
        val cid = newClientId()
        val id = store.add(NewMessage(conv, "out", data, "pending", kindForFilename(filename), filename, mimeType,
            clientMsgId = cid, mediaRef = up.filename))
        val msg = try {
            rest.postMessage(roomId, "", cid, media = up.filename)
        } catch (e: Exception) {
            store.setStatus(id, "failed")
            throw e
        }
        store.markSent(id, msg.id)
        return id
    }

    /** Байты вложения: из кэша или докачать и закэшировать. */
    suspend fun fetchMedia(messageId: Long): ByteArray {
        val row = store.get(messageId) ?: throw IllegalArgumentException("сообщение не найдено")
        row.body?.let { return it }
        val ref = row.mediaRef ?: throw IllegalStateException("нет вложения")
        return rest.downloadMedia(store.roomOf(row.conversationId), ref).also { store.setBody(messageId, it) }
    }

    private fun ingest(conv: Long, msg: RemoteMessage): Long? {
        if (store.findByWire(conv, msg.id) != null) return null
        msg.clientMsgId?.let { cid ->
            store.findOutByClientId(conv, cid)?.let { pending ->
                if (pending.wireSeq == null) store.markSent(pending.id, msg.id) // своё эхо
                return null
            }
        }
        if (ownUsername != null && msg.sender == ownUsername) {
            store.findUnconfirmedOutByBody(conv, msg.body.toByteArray())?.let {
                store.markSent(it.id, msg.id) // эхо без client_msg_id (веб-путь)
                return null
            }
        }
        val ts = epochMs(msg.createdAt)
        val id = if (msg.media != null) {
            val name = displayName(msg.media)
            store.add(NewMessage(conv, "in", null, "received", kindForFilename(msg.media), name,
                URLConnection.guessContentTypeFromName(name), msg.sender, msg.id, mediaRef = msg.media,
                replyAuthor = msg.reply?.sender, replySnippet = msg.reply?.body, timestamp = ts))
        } else {
            store.add(NewMessage(conv, "in", msg.body.toByteArray(), "received", sender = msg.sender, wireSeq = msg.id,
                replyAuthor = msg.reply?.sender, replySnippet = msg.reply?.body, timestamp = ts))
        }
        onMessage(conv, id)
        return id
    }

    private fun newClientId() = UUID.randomUUID().toString().replace("-", "")
}

package ru.mobile.mys.central

import java.util.concurrent.TimeUnit
import kotlinx.coroutines.CancellationException
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.Job
import kotlinx.coroutines.SupervisorJob
import kotlinx.coroutines.cancel
import kotlinx.coroutines.cancelAndJoin
import kotlinx.coroutines.channels.BufferOverflow
import kotlinx.coroutines.delay
import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.launch
import kotlinx.coroutines.sync.Mutex
import kotlinx.coroutines.sync.withLock
import okhttp3.OkHttpClient

sealed interface CentralEvent {
    /** [messageId] 0 — «содержимое беседы изменилось» (правка/удаление). */
    class Message(val conversationId: Long, val messageId: Long) : CentralEvent
    /** synced | connected | disconnected | unauthorized */
    class State(val state: String) : CentralEvent
    class Error(val error: Throwable) : CentralEvent
}

/** Оркестратор «Центра» — порт mys_centralized/service.py на корутинах. */
class CentralService(
    private val store: CentralStore,
    private val http: OkHttpClient = OkHttpClient.Builder().pingInterval(20, TimeUnit.SECONDS).build(),
    private val roomPollMs: Long = 20_000,
) {
    private val scope = CoroutineScope(SupervisorJob() + Dispatchers.IO)
    /** Синк, live-кадры, опрос, создание/вход — по одному: иначе гонка дедупа и бесед. */
    private val syncLock = Mutex()
    private var rest: RestClient? = null
    private var sync: SyncEngine? = null
    private var live: List<Job> = emptyList()
    private var knownRooms: Set<Long> = emptySet()
    private val _events = MutableSharedFlow<CentralEvent>(extraBufferCapacity = 256, onBufferOverflow = BufferOverflow.DROP_OLDEST)

    val events: SharedFlow<CentralEvent> = _events

    @Volatile var session: Session? = null
        private set

    /** /ws прошёл ready (для тестов и индикатора связи). */
    @Volatile var liveConnected = false
        private set

    fun hasSavedSession() = Account.load(store) != null

    fun wipeOnLogout() = store.setting(Account.K_WIPE) == "1"

    fun setWipeOnLogout(value: Boolean) = store.setSetting(Account.K_WIPE, if (value) "1" else "0")

    suspend fun login(serverUrl: String, username: String, password: String, register: Boolean = false): Session {
        val r = RestClient(serverUrl, http = http)
        val s = if (register) r.register(username, password) else r.login(username, password)
        activate(r, s)
        return s
    }

    /** Восстановить сохранённую сессию; протухший токен — сессия стирается, AuthException наружу. */
    suspend fun resume(): Session? {
        val s = Account.load(store) ?: return null
        try {
            activate(RestClient(s.serverUrl, s.token, http), s)
        } catch (e: AuthException) {
            teardown()
            Account.clear(store)
            throw e
        }
        return s
    }

    suspend fun logout() {
        val r = rest
        teardown()
        runCatching { r?.logout() } // best-effort: токен на сервере
        Account.clear(store)
        if (wipeOnLogout()) Account.wipeCache(store)
    }

    suspend fun createRoom(name: String): Long = syncLock.withLock { engine().createRoom(name) }

    suspend fun joinRoom(code: String): Long = syncLock.withLock { engine().joinRoom(code) }

    fun roomCode(conversationId: Long): String? = store.setting(Account.CODE_PREFIX + store.roomOf(conversationId))

    /** Принудительный дотяг комнат и истории. */
    suspend fun resync() { syncLock.withLock { engine().syncAll() } }

    suspend fun send(conversationId: Long, text: String, reply: ReplyRef? = null) = engine().send(conversationId, text, reply)
    suspend fun edit(messageId: Long, body: String) = engine().edit(messageId, body)
    suspend fun delete(messageId: Long) = engine().delete(messageId)
    suspend fun sendFile(conversationId: Long, filename: String, mimeType: String, data: ByteArray) =
        engine().sendFile(conversationId, filename, mimeType, data)
    suspend fun fetchMedia(messageId: Long) = engine().fetchMedia(messageId)

    suspend fun stop() {
        teardown()
        scope.cancel()
    }

    private fun engine() = sync ?: throw IllegalStateException("нет активной сессии «Центра»")

    private suspend fun activate(r: RestClient, s: Session) {
        teardown()
        rest = r
        session = s
        Account.save(store, s)
        val engine = SyncEngine(store, r, onMessage = { c, m -> _events.tryEmit(CentralEvent.Message(c, m)) }, ownUsername = s.username)
        sync = engine
        knownRooms = syncLock.withLock { engine.syncAll() }.keys
        _events.tryEmit(CentralEvent.State("synced"))
        live = listOfNotNull(
            wsUrlFor(s.serverUrl)?.let { url -> scope.launch { wsLoop(url, s.token, engine) } },
            if (roomPollMs > 0) scope.launch { pollLoop(engine) } else null,
        )
    }

    private suspend fun wsLoop(url: String, token: String, engine: SyncEngine) {
        try {
            WsClient(url, token, http).events().collect { e ->
                when (e.optString("type")) {
                    "ready" -> {
                        // после каждого (пере)подключения добираем пропущенное за офлайн
                        runCatching { syncLock.withLock { engine.syncAll() } }
                        liveConnected = true
                        _events.tryEmit(CentralEvent.State("connected"))
                    }
                    "message" -> syncLock.withLock { engine.ingestWs(e) }
                    "message_edited" -> syncLock.withLock { engine.applyWsEdit(e) }?.let { _events.tryEmit(CentralEvent.Message(it, 0)) }
                    "message_deleted" -> syncLock.withLock { engine.applyWsDelete(e) }?.let { _events.tryEmit(CentralEvent.Message(it, 0)) }
                }
            }
        } catch (e: AuthException) {
            liveConnected = false
            _events.tryEmit(CentralEvent.State("unauthorized"))
            _events.tryEmit(CentralEvent.Error(e))
        } catch (e: CancellationException) {
            throw e
        } catch (e: Exception) {
            liveConnected = false
            _events.tryEmit(CentralEvent.State("disconnected"))
            _events.tryEmit(CentralEvent.Error(e))
        }
    }

    /** Сервер не пушит события о новых комнатах — дотягиваем список периодически. */
    private suspend fun pollLoop(engine: SyncEngine) {
        while (true) {
            delay(roomPollMs)
            try {
                val now = syncLock.withLock { engine.syncAll() }.keys
                if (now != knownRooms) {
                    knownRooms = now
                    _events.tryEmit(CentralEvent.State("synced"))
                }
            } catch (e: AuthException) {
                _events.tryEmit(CentralEvent.State("unauthorized"))
                _events.tryEmit(CentralEvent.Error(e))
                return
            } catch (e: CancellationException) {
                throw e
            } catch (e: Exception) {
                // сетевой сбой опроса — переживём до следующего тика
            }
        }
    }

    private suspend fun teardown() {
        live.forEach { it.cancelAndJoin() }
        live = emptyList()
        liveConnected = false
        sync = null
        rest = null
        session = null
    }
}

package ru.mobile.mys.storage

import ru.mobile.mys.central.CentralStore
import ru.mobile.mys.central.NewMessage
import ru.mobile.mys.central.StoredMessage

/** «Центр» в зашифрованном vault: беседы режима "centralized", room_id = строка серверного id. */
class VaultCentralStore(private val vault: Vault) : CentralStore {
    override fun setting(key: String) = vault.settings.get(key)
    override fun setSetting(key: String, value: String) = vault.settings.set(key, value)
    override fun deleteSetting(key: String) = vault.settings.delete(key)
    override fun deleteSettings(prefix: String) = vault.settings.deletePrefix(prefix)

    override fun conversationForRoom(roomId: Long, title: String?, create: Boolean): Long? {
        val key = roomId.toString().toByteArray()
        return vault.conversations.getByRoomId(key, MODE)?.id ?: if (create) vault.conversations.add(MODE, key, title) else null
    }

    override fun roomOf(conversationId: Long): Long =
        vault.conversations.get(conversationId)!!.roomId!!.decodeToString().toLong()

    override fun wipeConversations() = vault.conversations.list(MODE).forEach { vault.conversations.delete(it.id) }

    override fun add(m: NewMessage): Long = vault.messages.add(
        m.conversationId, m.direction, m.body, m.status, m.kind, m.filename, m.mimeType, m.sender,
        m.wireSeq, m.clientMsgId, m.mediaRef, m.replyAuthor, m.replySnippet, m.timestamp,
    )

    override fun get(messageId: Long) = vault.messages.get(messageId)?.stored()
    override fun findByWire(conversationId: Long, wireSeq: Long) = vault.messages.findByWire(conversationId, wireSeq)?.stored()
    override fun findOutByClientId(conversationId: Long, clientMsgId: String) = vault.messages.findOutByClientId(conversationId, clientMsgId)?.stored()
    override fun findUnconfirmedOutByBody(conversationId: Long, body: ByteArray) = vault.messages.findUnconfirmedOutByBody(conversationId, body)?.stored()
    override fun markSent(messageId: Long, wireSeq: Long) = vault.messages.markSent(messageId, wireSeq)
    override fun setStatus(messageId: Long, status: String) = vault.messages.setStatus(messageId, status)
    override fun setBody(messageId: Long, body: ByteArray) = vault.messages.setBody(messageId, body)
    override fun delete(messageId: Long) = vault.messages.delete(messageId)

    private fun Message.stored() = StoredMessage(id, conversationId, direction, body, status, wireSeq, mediaRef, kind, sender)

    private companion object {
        const val MODE = "centralized"
    }
}
